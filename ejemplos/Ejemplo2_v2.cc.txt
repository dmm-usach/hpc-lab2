extern "C" {
#include <bsp.h>
   }
  
   #include <stdio.h>
   #include <stdlib.h>
   #include <string.h>
  using namespace std;
  #include <vector>
  #include <iostream>
 
  int N;
 
  void bsp_main()
  {
      // ojo: bsp_begin() debe ir primero.
      bsp_begin( bsp_nprocs() );
      int i, p, // numero de procesadores.
      pid, // indentificador del proceso (procesador)
      nm, // numero de mensajes que recibe cada procesador.
      nb,  // total de bytes en la cola de mensajes.
      tag, // identificador (header) de mensajes (no lo usamos).
      sztag, // tamaño del indentificador.
      total; // tamaño en bytes del siguiente mensaje a ser sacado
            // de la cola de nuevos mensajes en el procesador.
      tag = 0;
      sztag= sizeof(int);
      bsp_set_tagsize(&sztag);
      p= bsp_nprocs();
      pid = bsp_pid();

      int *dato = new int[N];
      if (pid==0) 
      {    //dato= 4;
           for (int i=0;i<N;i++)
              dato[i]= rand()%10;
 
           for(i=0;i<p;i++)
           { bsp_send(i, &tag, dato, sizeof(int)*N); }
      }
      bsp_sync();
      // lee cuantos mensajes llegaron (nm)
      // y cual es la cantidad total de bytes recibidos
      bsp_qsize( &nm, &nb);
      char pp[1024];
      // numero de mensajes recibidos.
      for(i=0; i<nm; i++)
      {
         bsp_get_tag(&total,&tag);
         bsp_move(pp,total);
         //memcpy(&dato[0],pp,sizeof(int)*N);
         memcpy(dato,pp,sizeof(int)*N);
      }
 
 
      for (int i=0; i<N;i++)
          printf("pid %d recibio dato %d\n",pid,dato[i]);
 
      bsp_end();
  }
 
  int main (int argc, char *argv[])
  {
          void bsp_main();
          if (argc <2)
          {
            cout<<"Error: Fatla ingresar parametro N"<<endl;
            return 0;
          }
          N = atoi(argv[1]);
 
          printf("Numero de elementos en el vector=%d\n",N);
          bsp_init(bsp_main, argc, argv);
          bsp_main();
          return 0;
}
